`src/sim/types.ts`: `FloorSegment` gains `/** Px/s along x that carries whoever stands on it: an escalator. */ belt?: number;`; `HeroState` gains `/** Seconds one-way platforms let him through after ↓ + jump. */ dropT: number;`.

`src/sim/physics.ts`:

```ts
export type Body = { x: number; y: number; w: number; h: number; vx: number; vy: number; onGround: boolean; dropT?: number };

/** The floor segment he stands on at x: the one `floorAt` would pick. */
export function floorUnder(stage: Stage, x: number, reach = -Infinity): FloorSegment | null {
  let best: FloorSegment | null = null;
  let bestY = Infinity;
  for (const f of stage.floor) {
    if (x < f.x0 - LEDGE || x > f.x1 + LEDGE) continue;
    const k = f.x1 > f.x0 ? Math.min(1, Math.max(0, (x - f.x0) / (f.x1 - f.x0))) : 0;
    const y = f.y0 + (f.y1 - f.y0) * k;
    if (y < reach || y >= bestY) continue;
    best = f;
    bestY = y;
  }
  return best;
}
```

`floorAt` becomes `const f = floorUnder(stage, x, reach); return f ? <y on f at x> : null;` (same result). In `moveBody(body, stage, dt, onBump?)`: in the vertical solid pass, the `else` branch (moving up into a solid) calls `onBump?.(s)` before `body.y = s.y + s.h + body.h`; the platform loop skips every platform while `(body.dropT ?? 0) > 0`.

`src/sim/hero.ts`:
- `createHero` sets `dropT: 0`.
- At the top of `stepHero`: `h.dropT = Math.max(0, h.dropT - dt);`.
- Before the jump: if `pressed.jump && held.down && h.onGround && !locked && onPlatform(h, stage)` then `h.dropT = 0.25; h.onGround = false; h.buffer = 0; h.crouch = false; h.h = HERO_H;` and skip the jump (a drop, not a jump), where `onPlatform` is true when no floor segment is within 1 px under his feet and a platform is (`stage.platforms.some(p => h.x >= p.x0 - LEDGE && h.x <= p.x1 + LEDGE && Math.abs(p.y - h.y) < 1)` and `floorAt(stage, h.x, h.y - 1)` is not `h.y`).
- Before `moveBody`: when on the ground, `const seg = floorUnder(stage, h.x, h.y - SNAP); if (seg?.belt) h.x += seg.belt * dt;`.
- `moveBody(h, stage, dt, (s) => events.push({ type: "bump", x: Math.min(s.x + s.w, Math.max(s.x, h.x)), y: s.y + s.h }))`: the point is clamped into the solid, so the Game can tell which crate his head met even when his middle is past its edge.

```ts
/**
 * Where pickups may go (the shipped rule, scaled): on level floor and on shelves, one anchor per
 * ANCHOR_STEP, never over a gap or a slope, inside a solid, in a doorway, on a cart, under a crate
 * or on a punch clock. A pickup's middle sits PICKUP_SIZE above the surface.
 */
export function rewardAnchors(level: Level): Point[] {
  const out: Point[] = [];
  const blocked = (x: number, y: number) =>
    level.solids.some((s) => x > s.x - 40 && x < s.x + s.w + 40 && y > s.y - PICKUP_SIZE) ||
    level.doors.some((d) => Math.abs(x - d.x) < CLEAR.door) ||
    level.carts.some((c) => Math.abs(x - c) < CLEAR.cart) ||
    level.crates.some((c) => Math.abs(x - c.x) < CLEAR.crate && y > c.y) ||
    level.checkpoints.some((k) => Math.abs(x - k.x) < CLEAR.checkpoint);
  for (const f of level.floor) {
    if (f.y0 !== f.y1 || f.belt) continue;
    for (let x = f.x0 + ANCHOR_STEP; x < f.x1 - ANCHOR_STEP / 2; x += ANCHOR_STEP) {
      const y = f.y0 - PICKUP_SIZE;
      if (x < level.spawn.x + 300 || x > level.length - 300 || blocked(x, y)) continue;
      out.push({ x, y });
    }
  }

    for (let x = p.x0 + ANCHOR_STEP / 2; x < p.x1 - 40; x += ANCHOR_STEP) {
      const y = p.y - PICKUP_SIZE;
      if (!blocked(x, y)) out.push({ x, y });
```

```ts
import { rewardAnchors, type Level, type RewardRoute } from "./level";
import type { Body } from "./physics";
import type { Point } from "./types";

/** The shipped generator (mulberry32), so a seed lays a run out the same way every time. */
export function seededRandom(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

const inRoute = (a: Point, r: RewardRoute) => a.x >= r.x0 && a.x <= r.x1 && a.y >= r.y0 && a.y <= r.y1;

/**
 * The shipped placement: shuffle the anchors, then Buc-ee's, nuts and cans in that order. Each of
 * the level's routes first takes one of each kind from its own shuffled anchors, so no seed leaves
 * a route without one; the totals stay the level's.
 */
export function seedRewards(level: Level, seed: number): Pickup[] {
  const anchors = rewardAnchors(level);
  const rng = seededRandom(seed);
  for (let i = anchors.length - 1; i > 0; i--) {
    const j = Math.floor(rng() * (i + 1));
    [anchors[i], anchors[j]] = [anchors[j], anchors[i]];
  }
  const n = Math.min(anchors.length, level.rewards.count);
  const left: Record<PickupKind, number> = { bucees: level.rewards.bucees, nuts: level.rewards.nuts, can: n - level.rewards.bucees - level.rewards.nuts, pepper: 0 };
  const out: Pickup[] = [];
  const place = (kind: PickupKind, a: Point) => {
    out.push({ id: out.length + 1, kind, x: a.x, y: a.y, taken: false });
    left[kind]--;
  };
  for (const r of level.routes)
    for (const kind of ["bucees", "nuts", "can"] as const) {
      const i = anchors.findIndex((a) => inRoute(a, r));
      if (i >= 0 && left[kind] > 0) place(kind, anchors.splice(i, 1)[0]);
    }
  for (const kind of ["bucees", "nuts", "can"] as const) while (left[kind] > 0 && anchors.length) place(kind, anchors.shift()!);
  return out;
}
```

```ts
import type { GameEvent } from "./events";
import type { CrateKind, Level } from "./level";
import type { Pickup } from "./rewards";
import type { Rect } from "./types";

/** The shipped 16 px crates at the new scale; a case's cans; an item's rise and slide. */
export const CRATE = { size: 110, bumpLock: 0.16, caseCans: 5, emerge: 0.5, slide: 286 } as const;

export type Crate = { id: string; kind: CrateKind; x: number; y: number; used: boolean; broken: boolean; cans: number; lock: number; hitBy: object | null };
export type CrateHit = "bump" | "mallet" | "burst";

export const cratesOf = (level: Level): Crate[] =>
  level.crates.map((c) => ({ ...c, used: false, broken: false, cans: 0, lock: 0, hitBy: null }));
export const crateRect = (c: Crate): Rect => ({ x: c.x - CRATE.size / 2, y: c.y - CRATE.size, w: CRATE.size, h: CRATE.size });
/** Every crate stays solid once spent; a broken case is gone. */
export const crateSolids = (crates: Crate[]): Rect[] => crates.filter((c) => !c.broken).map(crateRect);

/** One hit on a crate (the shipped hitBlock): what comes out, whether it is spent, whether a case breaks. */
export function hitCrate(
  c: Crate,
  how: CrateHit,
  ctx: { patience: number; star: boolean; hot: boolean },
  out: { pickups: Pickup[]; events: GameEvent[]; nextId: () => number },
): void {
  if (c.broken || c.lock > 0) return;
  const top = { x: c.x, y: c.y - CRATE.size };
  if (c.kind === "case") {
    c.lock = CRATE.bumpLock;
    c.cans++;
    out.events.push({ type: "canPop", x: top.x, y: top.y });
    const breaks = c.cans >= CRATE.caseCans || how !== "bump" || ctx.star || ctx.patience >= 2;
    if (breaks) {
      c.broken = true;
      out.events.push({ type: "caseBreak", x: c.x, y: c.y - CRATE.size / 2 });
    }
    return;
  }
  if (c.used) return;
  c.used = true;
  c.lock = CRATE.bumpLock;
  out.events.push({ type: "crate", id: c.id, kind: c.kind, x: c.x, y: c.y });
  if (c.kind === "can") {
    out.events.push({ type: "canPop", x: top.x, y: top.y });
    return;
  }
  const kind = c.kind === "bucees" ? "bucees" : ctx.hot ? "nuts" : "pepper";
  // It starts inside the crate (bottoms level) and rises out of the top.
  out.pickups.push({
    id: out.nextId(), kind, x: c.x, y: c.y - 35, taken: false,
    body: { x: c.x, y: c.y, w: 60, h: 70, vx: 0, vy: 0, onGround: false, emerge: CRATE.emerge, dir: 1 },
  });
}

/** Locks run down; an item out of a crate rises, then slides away under gravity. */
export function stepCrates(crates: Crate[], dt: number): void {
  for (const c of crates) c.lock = Math.max(0, c.lock - dt);
}
```

```ts
import { CRATE } from "./crates";
import { GRAVITY_DOWN } from "./constants";
import type { GameEvent } from "./events";
import { moveBody } from "./physics";
import type { HeroState, Stage } from "./types";

export function stepPickups(pickups: Pickup[], stage: Stage, dt: number): void {
  for (const p of pickups) {
    const b = p.body;
    if (!b || p.taken) continue;
    if (b.emerge > 0) {
      b.emerge = Math.max(0, b.emerge - dt);
      b.y -= (CRATE.size / CRATE.emerge) * dt;
    } else {
      b.vx = CRATE.slide * b.dir;
      b.vy += GRAVITY_DOWN * dt;
      moveBody(b, stage, dt);
      // A wall stops it (moveBody zeroes vx); it heads back the other way.
      if (b.vx === 0) b.dir = -b.dir;
    }
    p.x = b.x;
    p.y = b.y - b.h / 2;
  }
}

/** Everything his body overlaps is his (once an item is fully out of its crate). */
export function collectPickups(h: Pick<HeroState, "x" | "y" | "w" | "h">, pickups: Pickup[], events: GameEvent[]): PickupKind[] {
  const got: PickupKind[] = [];
  const half = 35;
  for (const p of pickups) {
    if (p.taken || (p.body && p.body.emerge > 0)) continue;
    if (p.x + half < h.x - h.w / 2 || p.x - half > h.x + h.w / 2 || p.y + half < h.y - h.h || p.y - half > h.y) continue;
    p.taken = true;
    got.push(p.kind);
    events.push({ type: "pickup", kind: p.kind, x: p.x, y: p.y });
  }
  return got;
}
```

```ts
/** The shipped run's numbers (src/game/engine.ts), with the clock and the tally in seconds. */
export const RUN = { … as in Interfaces … } as const;

export type Run = { seed: number; score: number; cans: number; lives: number; time: number; star: number; checkpoint: string | null };

export const createRun = (seed: number): Run => ({ seed, score: 0, cans: 0, lives: RUN.lives, time: 0, star: 0, checkpoint: null });

export function addScore(r: Run, n: number): void {
  r.score = Math.min(RUN.maxScore, r.score + n);
}

/** Cans: points, a cooler head, and a life for every hundred. Returns the heat they take off. */
export function addCans(r: Run, n: number, events: GameEvent[]): number {
  addScore(r, RUN.canPoints * n);
  r.cans += n;
  while (r.cans >= RUN.cansPerLife) {
    r.cans -= RUN.cansPerLife;
    r.lives++;
    events.push({ type: "oneUp" });
  }
  return RUN.canCool * n;
}
```

```ts
import { emptyInput, type Input } from "./input";
import { stageOf, type Level } from "./level";
import { floorAt } from "./physics";
import type { HeroState } from "./types";

/** A scripted player for the traversal test: run right, hop what is in the way, slide the passes, pick a route at the stairs. */
export function drive(level: Level, h: HeroState, route: "floor" | "balcony"): Input {
  const i = emptyInput();
  i.held.right = i.held.run = true;
  const stage = stageOf(level);
  const ahead = (d: number) => floorAt(stage, h.x + d, h.y - 60);
  const wall = level.solids.some((s) => s.x - h.x > 0 && s.x - h.x < 260 && s.y < h.y - 10 && s.y + s.h > h.y - 290);
  const gap = h.onGround && ahead(220) === null;
  const landing = route === "balcony" && level.platforms.some((p) => p.kind === "landing" && p.x0 - h.x > 0 && p.x0 - h.x < 200) && h.y === 0;
  const pass = level.passes.some((p) => p.x - h.x > -p.w && p.x - h.x < 300);
  if (pass) i.held.down = true;
  if ((wall || gap || landing) && h.onGround && !pass) i.held.jump = i.pressed.jump = true;
  else if (!h.onGround) i.held.jump = true;
  return i;
}
```

```ts
/** The escalator's steps this moment: one every 60 px of its run, sliding along at the belt's speed. */
export function escalatorSteps(seg: FloorSegment & { belt?: number }, t: number): Point[] {
  const run = seg.x1 - seg.x0;
  const gap = 60;
  const shift = ((((seg.belt ?? 0) * t) % gap) + gap) % gap;
  const out: Point[] = [];
  for (let x = seg.x0 + shift; x <= seg.x1; x += gap) out.push({ x, y: seg.y0 + ((seg.y1 - seg.y0) * (x - seg.x0)) / run });
  return out;
}
```
