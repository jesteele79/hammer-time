```ts
import type { GameEvent } from "./events";

export type Feel = {
  hitstop: number;
  trauma: number;
  slowmo?: { scale: number; seconds: number };
  zoom?: { amount: number; seconds: number };
};

/** What each event does to time and the camera (spec §5.4, §6.4, §9). */
export function feelOf(e: GameEvent): Feel | null {
  switch (e.type) {
    case "hit":
      if (e.by === "mallet") return { hitstop: 0.06, trauma: 0.35 };
      if (e.by === "snot") return { hitstop: 0.03, trauma: 0.12 };
      return null;
    case "burst":
      return { hitstop: 0.08, trauma: 0.8, slowmo: { scale: 0.25, seconds: 0.25 }, zoom: { amount: 1.12, seconds: 0.4 } };
    case "superStart":
      return { hitstop: 0, trauma: 0.15 };
    case "fire":
      return { hitstop: 0, trauma: 0.08 };
    default:
      return null;
  }
}
```

`src/sim/clock.ts`:

```ts
import type { Feel } from "./feel";

/** Hit-stop and slow motion, both counted in wall seconds. */
export type Clock = { hitstop: number; slowmo: number; scale: number };

export const createClock = (): Clock => ({ hitstop: 0, slowmo: 0, scale: 1 });

/** The longer hit-stop wins, never the sum; slow motion keeps the deeper scale and the longer time. */
export function applyFeel(c: Clock, f: Feel): void {
  c.hitstop = Math.max(c.hitstop, f.hitstop);
  if (f.slowmo) {
    c.scale = c.slowmo > 0 ? Math.min(c.scale, f.slowmo.scale) : f.slowmo.scale;
    c.slowmo = Math.max(c.slowmo, f.slowmo.seconds);
  }
}

/** The sim seconds `dt` wall seconds are worth. Slow motion keeps counting through a hit-stop. */
export function tickClock(c: Clock, dt: number): number {
  let left = dt;
  let sim = 0;
  if (c.hitstop > 0) {
    const used = Math.min(left, c.hitstop);
    c.hitstop -= used;
    c.slowmo = Math.max(0, c.slowmo - used);
    left -= used;
  }
  if (left > 0 && c.slowmo > 0) {
    const used = Math.min(left, c.slowmo);
    sim += used * c.scale;
    c.slowmo -= used;
    left -= used;
  }
  sim += left;
  if (c.slowmo <= 0) c.scale = 1;
  return sim;
}
```

```ts
export const MOVE_SECONDS: Record<MoveKind, number> = { swing: 0.42, super: 0.66, snot: 0.45, wipe: 0.95 };
/** Where a swing can land, as a fraction of the move. */
export const SWING_ACTIVE = [0.33, 0.72] as const;
/** The strike over the head, sampled for contact but never drawn. */
export const SWING_CUT = [0.33, 0.36] as const;
export const SUPER_CUT = [0.42, 0.46] as const;
export const SNOT_FIRE_U = 0.45;
export const WIPE_SCRUB = [0.12, 0.62] as const;
export const WIPE_FLICK = [0.62, 0.78] as const;
export const WIPE_SHINE = 0.84;
/** Heat per second running and resting, per action, the wipe's drain, and the thresholds. */
export const HEAT = { run: 6, rest: 3, jump: 12, swing: 3, snot: 5, wipe: 240, tired: 80, boil: 99.5, cool: 70 } as const;
export const SINUS_MAX = 3;
export const SINUS_REFILL = 1.5;
export const FIRE_COOLDOWN = 0.5;
export const SNOT_ALIVE = 3;
export const CHARGE_MAX = 10;
/** The shipped hop and plunge (260 and 300 px/s at 50 px tall), scaled by the run speeds. */
export const SUPER_HOP = 1020;
export const SUPER_PLUNGE = 1215;
/** Caps while tired, as fractions of the run and walk speeds. */
export const TIRED = { run: 0.73, walk: 0.67 } as const;

const newMove = (kind: MoveKind): Move => ({ kind, t: 0, fired: false, burst: false, lastU: -1, stroke: -1, victims: [] });
export const moveU = (m: Move) => Math.min(1, m.t / MOVE_SECONDS[m.kind]);

/** Starts at most one move, by the shipped priority: towel, snot, super, swing. Nothing starts crouched or sliding. */
export function startMove(h: HeroState, input: Input, alive: number, events: GameEvent[]): void {
  if (h.move || h.crouch) return;
  const p = input.pressed;
  if (p.towel && h.heat > 0 && h.onGround) {
    h.move = newMove("wipe");
    h.vx *= 0.2;
    events.push({ type: "wipeStart" });
  } else if (p.fire && h.sinus >= 1 && h.fireCd <= 0 && alive < SNOT_ALIVE) {
    h.move = newMove("snot");
    h.fireCd = FIRE_COOLDOWN;
    h.sinus -= 1;
    events.push({ type: "sniff" });
  } else if (p.super && h.charge >= CHARGE_MAX) {
    h.move = newMove("super");
    h.charge = 0;
    h.vy = -SUPER_HOP;
    h.onGround = false;
    events.push({ type: "superStart" });
  } else if (p.hammer) {
    h.move = newMove("swing");
    h.heat = Math.min(100, h.heat + HEAT.swing);
    events.push({ type: "swing" });
  }
}

/** Before movement: the super plunges once the load is done, until it lands. */
export function steerMove(h: HeroState): void {
  const m = h.move;
  if (m?.kind === "super" && !m.burst && moveU(m) >= SUPER_CUT[0] && !h.onGround) h.vy = Math.max(h.vy, SUPER_PLUNGE);
}

/** After movement: the clock, the wipe's strokes, the super's landing flag, the end. */
export function advanceMove(h: HeroState, dt: number, events: GameEvent[]): void {
  const m = h.move;
  if (!m) return;
  m.t += dt;
  const u = moveU(m);
  if (m.kind === "super" && !m.burst && u >= SUPER_CUT[0] && (h.onGround || u >= 0.92)) m.burst = true;
  if (m.kind === "wipe") {
    const stroke =
      u < WIPE_SCRUB[0] ? -1 : u < WIPE_SCRUB[1] ? Math.floor((u - WIPE_SCRUB[0]) / 0.1) : u < WIPE_FLICK[0] + 0.04 ? 5 : u < WIPE_SHINE ? 6 : 7;
    if (stroke !== m.stroke) {
      m.stroke = stroke;
      if (stroke >= 0 && stroke < 5) events.push({ type: "stroke", n: stroke });
      else if (stroke === 6) events.push({ type: "flick" });
      else if (stroke === 7) events.push({ type: "shine" });
    }
  }
  if (m.t >= MOVE_SECONDS[m.kind]) h.move = null;
}

/** Heat, the sinus and the snot cooldown, every step. */
export function stepHeat(h: HeroState, input: Input, dt: number, events: GameEvent[]): void {
  if (h.move?.kind === "wipe") h.heat = Math.max(0, h.heat - HEAT.wipe * dt);
  else {
    const running = input.held.run && Math.abs(h.vx) > WALK;
    h.heat = Math.min(100, Math.max(0, h.heat + (running ? HEAT.run : -HEAT.rest) * dt));
  }
  if (h.heat >= HEAT.boil && !h.boiled) {
    h.boiled = true;
    events.push({ type: "boiling" });
  } else if (h.heat < HEAT.cool) h.boiled = false;
  h.sinus = Math.min(SINUS_MAX, h.sinus + dt / SINUS_REFILL);
  h.fireCd = Math.max(0, h.fireCd - dt);
}
```

```ts
  events: GameEvent[];

  const world: World = {
    stage,
    hero: createHero(spawn, opts),
    time: 0,
    events: [],

      const h = world.hero;

      stepHero(h, world.stage, input, STEP, events);
      advanceMove(h, STEP, events);
      stepHeat(h, input, STEP, events);

      return { time: world.time, hero: { ...h, move: h.move && { ...h.move, victims: [...h.move.victims] } } };
    },
  };
```

```ts
import { JOINTS, type JointName, type Point } from "./skeleton";

const DEG = 180 / Math.PI;
const ang = (a: Point, b: Point) => Math.atan2(b.y - a.y, b.x - a.x) * DEG;
const norm = (d: number) => ((((d + 180) % 360) + 360) % 360) - 180;
const CHAINS: Record<"near" | "far", [JointName, JointName, JointName, JointName]> = {
  near: ["shoulderNear", "elbowNear", "wristNear", "knuckleNear"],
  far: ["shoulderFar", "elbowFar", "wristFar", "handFar"],
};

/**
 * Shoulder, elbow and wrist rotations (degrees, the chest's frame, facing right) that put the
 * wrist on `target` (rest units) and turn the hand to `handDeg`. The elbow sits on the
 * counter-clockwise side of the shoulder-to-wrist line for `bend` 1, clockwise for −1. A target
 * out of reach is pulled in along that line.
 */
export function armTo(arm: "near" | "far", target: Point, handDeg: number, bend: 1 | -1) {
  const [s, e, w, k] = CHAINS[arm];
  const S = JOINTS[s].rest, E0 = JOINTS[e].rest, W0 = JOINTS[w].rest, K0 = JOINTS[k].rest;
  const a = Math.hypot(E0.x - S.x, E0.y - S.y);
  const b = Math.hypot(W0.x - E0.x, W0.y - E0.y);
  const reach = Math.hypot(target.x - S.x, target.y - S.y);
  const d = Math.min(a + b - 1e-6, Math.max(Math.abs(a - b) + 1e-6, reach));
  const base = Math.atan2(target.y - S.y, target.x - S.x);
  const off = Math.acos((a * a + d * d - b * b) / (2 * a * d));
  const upperRad = base + bend * off;
  const E = { x: S.x + a * Math.cos(upperRad), y: S.y + a * Math.sin(upperRad) };
  const T = { x: S.x + d * Math.cos(base), y: S.y + d * Math.sin(base) };
  const shoulder = norm(upperRad * DEG - ang(S, E0));
  const elbow = norm(ang(E, T) - ang(E0, W0) - shoulder);
  const wrist = norm(handDeg - ang(W0, K0) - shoulder - elbow);
  return { shoulder, elbow, wrist };
}
```

```ts
export function knock(t: Target, dir: 1 | -1, speed: number, lift: number): void {
  t.vx = dir * speed;
  t.vy = -lift;
  t.onGround = false;
  t.flash = 0.06;
  t.spin = dir * 6;
}

export function stepTargets(targets: Target[], stage: Stage, dt: number): void {
  for (const t of targets) {
    t.flash = Math.max(0, t.flash - dt);
    if (!t.onGround) t.vy += GRAVITY_DOWN * dt;
    else {
      const f = CART.friction * dt;
      t.vx = Math.abs(t.vx) <= f ? 0 : t.vx - Math.sign(t.vx) * f;
      t.spin *= Math.max(0, 1 - 8 * dt);
    }
    moveBody(t, stage, dt);
  }
}
```

```ts
/** The mallet head's reach for contact, world px. */
export const MALLET_R = 36;
/** The super burst's radius (54 px at 50 tall, scaled by HERO_HEIGHT). */
export const BURST_R = 367;

/** A head of radius r moving from a to b touches the rect anywhere along the way (slab test on the grown rect). */
export function sweptHeadHits(a: Point, b: Point, r: number, rect: Rect): boolean {
  let lo = 0;
  let hi = 1;
  const dx = b.x - a.x;
  const dy = b.y - a.y;
  for (const [p, q] of [
    [-dx, a.x - rect.x + r],
    [dx, rect.x + rect.w + r - a.x],
    [-dy, a.y - rect.y + r],
    [dy, rect.y + rect.h + r - a.y],
  ]) {
    if (Math.abs(p) < 1e-9) {
      if (q < 0) return false;
      continue;
    }
    const t = q / p;
    if (p < 0) lo = Math.max(lo, t);
    else hi = Math.min(hi, t);
    if (lo > hi) return false;
  }
  return true;
}

/** Where the mallet head is at move fraction u, in world px, from the same rig the renderer draws. */
export function malletAt(h: Pick<HeroState, "x" | "y" | "facing"> & { move: Pick<Move, "kind"> | null }, u: number): Point {
  const kind = h.move?.kind ?? "swing";
  const j = solve(ACTIONS[kind](u), h.facing, HEAD_SCALE).joints.malletHead;
  return { x: h.x + j.x, y: h.y + j.y };
}

const rectOf = (t: Target): Rect => ({ x: t.x - t.w / 2, y: t.y - t.h, w: t.w, h: t.h });

/** The swing's contact: the arc since the last check, sampled finely, against every cart not yet hit. */
export function swingContact(h: HeroState, targets: Target[], events: GameEvent[]): void {
  const m = h.move;
  if (m?.kind !== "swing") return;
  const u = moveU(m);
  const last = m.lastU < 0 || m.lastU > u ? u : m.lastU;
  m.lastU = u;
  const [lo, hi] = SWING_ACTIVE;
  if (u < lo || last > hi) return;
  const from = Math.max(last, lo);
  const to = Math.min(u, hi);
  const n = Math.max(1, Math.ceil((to - from) / 0.008));
  const path: Point[] = [];
  for (let k = 0; k <= n; k++) path.push(malletAt(h, from + ((to - from) * k) / n));
  // Only the arc in front of him lands: the windmill passes over his head and behind him on the way.
  const ahead = (p: Point) => (p.x - h.x) * h.facing > 0;
  for (const t of targets) {
    if (m.victims.includes(t.id)) continue;
    const r = rectOf(t);
    for (let k = 1; k < path.length; k++)
      if (ahead(path[k]) && sweptHeadHits(path[k - 1], path[k], MALLET_R, r)) {
        m.victims.push(t.id);
        knock(t, h.facing, 1100, 600);
        events.push({ type: "hit", by: "mallet", x: path[k].x, y: path[k].y, target: t.id });
        h.charge = Math.min(10, h.charge + 1);
        break;
      }
  }
}

/** The slam: once, where the mallet meets the floor; every cart within the radius flies. */
export function superBurst(h: HeroState, targets: Target[], events: GameEvent[]): void {
  const m = h.move;
  if (m?.kind !== "super" || !m.burst || m.fired) return;
  m.fired = true;
  const head = malletAt(h, Math.max(moveU(m), 0.56));
  const at = { x: head.x, y: h.y };
  events.push({ type: "burst", x: at.x, y: at.y });
  for (const t of targets) {
    const cx = t.x;
    const cy = t.y - t.h / 2;
    if (Math.hypot(cx - at.x, cy - at.y) > BURST_R) continue;
    knock(t, cx >= at.x ? 1 : -1, 1100, 600);
    events.push({ type: "hit", by: "burst", x: cx, y: cy, target: t.id });
  }
}
```

```ts
  const y = h.y + s.nose.y;

  h.heat = Math.min(100, h.heat + HEAT.snot);
  events.push({ type: "fire", x, y, hot: h.hot });
}

export function stepSnot(w: { stage: Stage; snot: Snot[]; puddles: Puddle[]; targets: Target[] }, dt: number, events: GameEvent[]): void {
  for (const b of w.snot) {
    const a = { x: b.x, y: b.y };
    b.vy += SNOT.gravity * dt;
    b.x += b.vx * dt;
    b.y += b.vy * dt;
    b.life -= dt;
    const r = b.hot ? SNOT.hotR : SNOT.r;
    for (const t of w.targets) {
      if (b.life <= 0 || b.hit.includes(t.id)) continue;
      if (!sweptHeadHits(a, b, r, { x: t.x - t.w / 2, y: t.y - t.h, w: t.w, h: t.h })) continue;
      b.hit.push(t.id);
      knock(t, b.vx > 0 ? 1 : -1, 420, 180);
      events.push({ type: "hit", by: "snot", x: b.x, y: b.y, target: t.id });
      events.push({ type: "splat", x: b.x, y: b.y, floor: false, hot: b.hot });
      if (b.pierce > 0) b.pierce--;
      else b.life = 0;
    }
    const floor = floorAt(w.stage, b.x, a.y);
    if (b.life > 0 && floor !== null && b.y >= floor) {
      b.life = 0;
      w.puddles.push({ x: b.x, y: floor, w: b.hot ? SNOT.hotPuddle : SNOT.puddle, life: b.hot ? SNOT.hotPuddleLife : SNOT.puddleLife, max: b.hot ? SNOT.hotPuddleLife : SNOT.puddleLife });
      if (w.puddles.length > SNOT.puddles) w.puddles.shift();
      events.push({ type: "splat", x: b.x, y: floor, floor: true, hot: b.hot });
    }
  }
  w.snot = w.snot.filter((b) => b.life > 0);
  for (const p of w.puddles) p.life -= dt;
  w.puddles = w.puddles.filter((p) => p.life > 0);
}
```

```ts
import { MeshRope, Point as PxPoint, type Container, type Texture } from "pixi.js";
import type { Point } from "../sim/types";

/**
 * The towel as a hanging chain: the first point is the hand, the rest follow at `len` apart
 * under gravity with damping, and no point goes past `back` (the head's back edge in front
 * of which the towel would cover the face): the towel passes behind the head, never in front.
 */
export function towelChain(anchor: Point, prev: Point[], n: number, len: number, back: number): Point[] {
  const pts: Point[] = [{ ...anchor }];
  for (let i = 1; i < n; i++) {
    const was = prev[i] ?? { x: anchor.x, y: anchor.y + len * i };
    const up = pts[i - 1];
    let x = was.x;
    let y = was.y + len * 0.35;
    const d = Math.hypot(x - up.x, y - up.y) || 1;
    x = up.x + ((x - up.x) / d) * len;
    y = up.y + ((y - up.y) / d) * len;
    pts.push({ x: Math.min(x, back), y });
  }
  return pts;
}

export class Towel {
  private readonly points: PxPoint[];
  private chain: Point[] = [];
  readonly rope: MeshRope;
  constructor(layer: Container, texture: Texture, readonly n = 10, readonly len = 22) {
    this.points = Array.from({ length: n }, () => new PxPoint());
    this.rope = new MeshRope({ texture, points: this.points, textureScale: 0 });
    this.rope.visible = false;
    layer.addChild(this.rope);
  }
  /** `anchor` is the far hand in the character's local px (null hides the towel); `back` is in the same frame. */
  update(anchor: Point | null, back: number): void {
    if (!anchor) {
      this.rope.visible = false;
      this.chain = [];
      return;
    }
    this.chain = towelChain(anchor, this.chain, this.n, this.len, back);
    this.chain.forEach((p, i) => this.points[i].set(p.x, p.y));
    this.rope.visible = true;
  }
}
```

```ts
import { AudioBus } from "../game/audio";
import type { GameEvent } from "../sim/events";

const CUES: Partial<Record<GameEvent["type"], (keyof AudioBus & string)[]>> = {
  jump: ["jump"], slide: ["slide"], swing: ["whoosh"], superStart: ["power"], burst: ["boom"],
  sniff: ["sniff"], fire: ["honk"], splat: ["splat"], wipeStart: ["cloth"], stroke: ["swish"],
  flick: ["snap"], shine: ["glint"], boiling: ["whistle"],
};

/** The shipped game's sound effects for the rebuild's events. Silent until the first key unlocks it. */
export function createSound() {
  const bus = new AudioBus();
  const played: Record<string, number> = {};
  const sound = {
    played,
    unlocked: false,
    unlock() {
      if (sound.unlocked) return;
      bus.unlock();
      sound.unlocked = !bus.failed;
    },
    play(e: GameEvent) {
      const names = e.type === "hit" ? (e.by === "mallet" ? ["bonk"] : e.by === "snot" ? ["splat"] : []) : (CUES[e.type] ?? []);
      for (const name of names) {
        played[name] = (played[name] ?? 0) + 1;
        if (sound.unlocked) (bus[name as keyof AudioBus] as () => void).call(bus);
      }
    },
    destroy() {
      bus.destroy();
    },
  };
  return sound;
}
```
