```ts
import type { GameEvent } from "../events";
import { smooth, type Pose } from "./pose";

/** The procedural layers over the clips (spec §6.3). */
export const LAYERS = {
  /** Degrees of lean per px/s² of horizontal acceleration (ground acceleration gives the cap), and how fast it follows. */
  leanPer: 9 / 4400,
  leanMax: 9,
  leanT: 0.08,
  breathHz: 0.4,
  breath: 0.015,
  /** The belly's spring: its rate (rad/s), the kick a landing at 2600 px/s gives and a jump's, and the cap. */
  bellyW: 18,
  bellyLand: 1.95,
  bellyJump: -1.2,
  bellyMax: 0.04,
  land: { x: 1.08, y: 0.92, t: 0.09 },
  jump: { x: 0.95, y: 1.06, t: 0.18 },
} as const;
const TERMINAL = 2600;

export type Layers = {
  lean: number;
  lastVx: number | null;
  t: number;
  belly: number;
  bellyV: number;
  squash: { x: number; y: number; t: number; age: number } | null;
};

export const createLayers = (): Layers => ({ lean: 0, lastVx: null, t: 0, belly: 0, bellyV: 0, squash: null });

/** One frame of sim time: the lean follows the acceleration, the breath runs, landings squash, jumps stretch, the belly springs. */
export function stepLayers(s: Layers, h: { vx: number; facing: 1 | -1 }, events: readonly GameEvent[], dt: number): void {
  if (dt <= 0) return;
  s.t += dt;
  const ax = s.lastVx === null ? 0 : (h.vx - s.lastVx) / dt;
  s.lastVx = h.vx;
  const want = Math.max(-LAYERS.leanMax, Math.min(LAYERS.leanMax, ax * h.facing * LAYERS.leanPer));
  s.lean += (want - s.lean) * (1 - Math.exp(-dt / LAYERS.leanT));
  for (const e of events) {
    if (e.type === "land") {
      s.squash = { ...LAYERS.land, age: 0 };
      s.bellyV += LAYERS.bellyLand * Math.min(1, e.speed / TERMINAL);
    } else if (e.type === "jump") {
      s.squash = { ...LAYERS.jump, age: 0 };
      s.bellyV += LAYERS.bellyJump;
    }
  }
  if (s.squash) {
    s.squash.age += dt;
    if (s.squash.age >= s.squash.t) s.squash = null;
  }
  // Critically damped, stepped exactly, so a long frame can't make it ring: x(t) = (x0 + (v0 + w x0) t) e^(−wt).
  const w = LAYERS.bellyW;
  const e = Math.exp(-w * dt);
  const b = s.bellyV + w * s.belly;
  const x = s.belly + b * dt;
  s.belly = x * e;
  s.bellyV = (b - w * x) * e;
  s.belly = Math.max(-LAYERS.bellyMax, Math.min(LAYERS.bellyMax, s.belly));
}

export function squashNow(s: Layers): { x: number; y: number } {
  if (!s.squash) return { x: 1, y: 1 };
  const k = 1 - smooth(Math.min(1, s.squash.age / s.squash.t));
  return { x: 1 + (s.squash.x - 1) * k, y: 1 + (s.squash.y - 1) * k };
}

export const breathNow = (s: Layers) => 1 + LAYERS.breath * Math.sin(2 * Math.PI * LAYERS.breathHz * s.t);

/** The acceleration lean, on the locomotion pose before any action is laid over it. */
export function addLean(p: Pose, s: Layers): Pose {
  return { ...p, lean: p.lean + s.lean };
}

/** The squash, the stretch, the breath and the belly, on the finished pose. */
export function addBody(p: Pose, s: Layers): Pose {
  const q = squashNow(s);
  return { ...p, squash: { x: p.squash.x * q.x, y: p.squash.y * q.y }, torso: p.torso * (breathNow(s) + s.belly) };
}
```

```ts
export type FloorProbe = (dx: number) => number | null;

/** A sole this close to its floor line (px) is standing on it. */
const PLANTED = 4;

const REACH = 48;

export function plantFeet(pose: Pose, facing: 1 | -1, headScale: number, plate: HeadPlate | undefined, mode: Ground, floor: FloorProbe): { pose: Pose; dy: number; gap: number } {
  const s0 = solve(pose, facing, headScale, plate);
  const dy0 = groundShift(s0, mode);
  if (mode !== "planted" && mode !== "push") return { pose, dy: dy0, gap: 0 };
  const j = s0.joints;
  const lift = FAR_FLOOR * s0.scale * s0.squash.y;
  const at = (n: JointName): Point => ({ x: j[n].x, y: j[n].y + dy0 });
  const feet = (["near", "far"] as Leg[]).flatMap((leg) => {
    const n = PARTS[leg];
    const line = leg === "far" ? -lift : 0;
    const heel = at(n.heel);
    const toe = at(n.toe);
    const pivot = heel.y >= toe.y ? heel : toe;
    const d = floor(pivot.x);
    const fh = floor(heel.x);
    const ft = floor(toe.x);
    if (d === null || fh === null || ft === null || [d, fh, ft].some((v) => Math.abs(v) > REACH)) return [];
    // The floor's own angle under the sole, whichever way he faces.
    const theta = Math.atan((ft - fh) / (toe.x - heel.x));
    return [{ leg, line, pivot, d, theta, standing: pivot.y >= line - PLANTED }];
  });
  const standing = feet.filter((f) => f.standing);
  const shift = standing.length ? Math.max(...standing.map((f) => f.d)) : 0;
  const moved = feet.filter((f) => f.standing || f.pivot.y + shift > f.line + f.d);
  if (moved.every((f) => Math.abs(f.d - shift) < 1e-9 && Math.abs(f.theta) < 1e-9)) return { pose, dy: dy0 + shift, gap: 0 };

  const out = structuredClone(pose);
  const rest = (p: Point): Point => ({
    x: p.x / (s0.scale * facing * s0.squash.x) - pose.root.x,
    y: -p.y / (s0.scale * s0.squash.y) - pose.root.y,
  });
  for (const f of moved) {
    const n = PARTS[f.leg];
    const c = Math.cos(f.theta);
    const s = Math.sin(f.theta);
    const to = { x: f.pivot.x, y: f.line + f.d };
    // The foot turns to the slope about the point it stands on and moves with it onto the floor.
    const place = (q: Point): Point => ({
      x: to.x + c * (q.x - f.pivot.x) - s * (q.y - f.pivot.y),
      y: to.y + s * (q.x - f.pivot.x) + c * (q.y - f.pivot.y),
    });
    const heelTo = place(at(n.heel));
    const toeTo = place(at(n.toe));
    let lift = 0;
    for (let k = 0; k < SOLE_SAMPLES; k++) {
      const t = k / (SOLE_SAMPLES - 1);
      const q = { x: heelTo.x + (toeTo.x - heelTo.x) * t, y: heelTo.y + (toeTo.y - heelTo.y) * t };
      const fq = floor(q.x);
      if (fq !== null) lift = Math.max(lift, q.y - (f.line + fq));
    }
    // Back to the solve's frame: the whole body is drawn `dy0 + shift` lower; the foot rises clear by `lift`.
    const back = (q: Point): Point => rest({ x: q.x, y: q.y - lift - dy0 - shift });
    const ankle = back(place(at(n.ankle)));
    const toe = back(toeTo);
    const hip = rest(j[n.hip]);
    const knee = rest(j[n.knee]);
    const was = rest(j[n.ankle]);
    // Keep the knee on the side it already bends to.
    const cross = (was.x - hip.x) * (knee.y - hip.y) - (was.y - hip.y) * (knee.x - hip.x);
    const r = legTo(f.leg, ankle, Math.atan2(toe.y - ankle.y, toe.x - ankle.x) * DEG, cross >= 0 ? 1 : -1);
    out.rot[n.hip] = r.hip;
    out.rot[n.knee] = r.knee;
    out.rot[n.ankle] = r.ankle;
  }
  const s1 = solve(out, facing, headScale, plate);
  const dy = dy0 + shift;
  let gap = 0;
  for (const f of moved) {
    const n = PARTS[f.leg];
    const off = [s1.joints[n.heel], s1.joints[n.toe]].map((q) => Math.abs(q.y + dy - (f.line + (floor(q.x) ?? f.d))));
    gap = Math.max(gap, Math.min(...off));
  }
  return { pose: out, dy, gap };
}
```

```ts
/** The shipped idle beats: the watch check at 2.5 s, the foot tap from 5 s, the harrumph at 8 s, then round again. */
export const IDLE = { watchAt: 2.5, tapAt: 5, harrumphAt: 8 } as const;
const IDLE_ACTS: ReadonlySet<MoveKind> = new Set<MoveKind>(["watch", "tap"]);
export const isIdleAct = (m: Move | null): boolean => m !== null && IDLE_ACTS.has(m.kind);

/** Standing around. Any key, any motion or any other move ends an idle act and starts the count again. */
export function stepIdle(h: HeroState, input: Input, dt: number, events: GameEvent[]): void {
  const busy =
    Object.values(input.held).some(Boolean) || !h.onGround || Math.abs(h.vx) > 1 || h.crouch || (h.move !== null && !isIdleAct(h.move));
  if (busy) {
    h.idleT = 0;
    h.bored = false;
    if (isIdleAct(h.move)) h.move = null;
    return;
  }
  const was = h.idleT;
  h.idleT += dt;
  const crossed = (t: number) => was < t && h.idleT >= t;
  if (crossed(IDLE.watchAt)) {
    h.move = newMove("watch");
    h.bored = true;
    events.push({ type: "watch" });
  }
  if (crossed(IDLE.tapAt)) {
    h.move = newMove("tap");
    events.push({ type: "tap" });
  }
  if (h.idleT >= IDLE.harrumphAt) {
    h.idleT = 0;
    events.push({ type: "harrumph" });
  }
}
```

```ts
import type { GameEvent } from "./events";
import { GRACE, newMove } from "./moves";

import type { HeroState, Stage } from "./types";

/** Hammer on damage (spec §9: 540 back, 500 up) and the hurt face's 0.6 s (§6.1). */
export const HURT = { back: 540, up: 500, face: 0.6 } as const;
/** The last pip: thrown higher, off his feet. */
export const KNOCKDOWN = { back: 540, up: 900 } as const;
/** The shipped patience: form + 1. */
export const PATIENCE_MAX = 3;

/**
 * A hit from something at `fromX`. In grace or already down it does nothing. Otherwise it costs a
 * patience pip, ends whatever he was doing, drops hot snot (the shipped game's lost form), turns
 * him to face what hit him and throws him back: hurt, or on the last pip the tumble.
 */
export function damage(h: HeroState, stage: Stage, fromX: number, events: GameEvent[]): boolean {
  if (h.invuln > 0 || h.move?.kind === "tumble") return false;
  const facing: 1 | -1 = fromX >= h.x ? 1 : -1;
  const dir: 1 | -1 = facing === 1 ? -1 : 1;
  const down = h.patience <= 1;
  h.facing = facing;
  h.patience = Math.max(0, h.patience - 1);
  h.hot = false;
  h.hurtT = HURT.face;
  h.invuln = GRACE;
  h.slideT = 0;
  h.buffer = 0;
  if (h.crouch && canStand(h, stage, HERO_H)) {
    h.crouch = false;
    h.h = HERO_H;
  }
  const kick = down ? KNOCKDOWN : HURT;
```

```ts
import { BitmapText, Container, Graphics } from "pixi.js";
import type { Point, Rect } from "../sim/types";
import { PALETTE } from "./palette";
import { overlaps } from "./words";

const EDGE = 16;
/** The shipped layout's offsets (src/game/thought.ts) at the new scale: above and behind his crown. */
const BEHIND = 68;
const ABOVE = 40;
const GAP = 20;
const TEXT = { size: 26, wrap: 400, padX: 26, padY: 18 } as const;
const PAPER = 0xfbf7ee;
const SPOKEN = 0xc44838;
const INK = PALETTE.outline;
const FADE = 0.2;

/**
 * Where a bubble goes (spec §5.7): above and behind his head; beside his face, behind him, when
 * that would cover it; below his chin as the last resort. The two puffs trail from the back of
 * his crown, or from under his chin when the bubble went below.
 */
export function bubbleLayout(size: { w: number; h: number }, crown: Point, facing: 1 | -1, face: Rect, view: Rect) {
  const fitX = (x: number) => Math.min(Math.max(x, view.x + EDGE), view.x + view.w - size.w - EDGE);
  const fitY = (y: number) => Math.min(Math.max(y, view.y + EDGE), view.y + view.h - size.h - EDGE);
  let box: Rect = { x: fitX(crown.x - facing * BEHIND - size.w / 2), y: fitY(crown.y - ABOVE - size.h), ...size };
  if (overlaps(box, face, 2)) box = { ...box, x: fitX(facing > 0 ? face.x - GAP - size.w : face.x + face.w + GAP) };
  if (overlaps(box, face, 2)) box = { ...box, y: fitY(face.y + face.h + GAP) };
  const below = box.y > face.y;
  const from = below
    ? { x: face.x + face.w / 2 - facing * (face.w / 2 + 14), y: face.y + face.h + 10 }
    : { x: crown.x - facing * 30, y: crown.y - 6 };
  const to = { x: Math.min(Math.max(from.x, box.x + 20), box.x + box.w - 20), y: below ? box.y : box.y + box.h };
  const puffs = [
    { t: 0.35, r: 9 },
    { t: 0.7, r: 14 },
  ].map(({ t, r }) => ({ x: from.x + (to.x - from.x) * t, y: from.y + (to.y - from.y) * t, r }));
  return { box, puffs };
}

/** Hammer's thoughts (clouds) and the caption of a recording he speaks (a box), one at a time. */
export class Bubbles {
  private readonly root = new Container({ label: "bubbles" });
  private readonly shape = new Graphics();
  private readonly words: BitmapText;
  private life = 0;
  private spoken = false;
  private current: string | null = null;

  constructor(layer: Container) {
    this.words = new BitmapText({ text: "", style: { fontFamily: "ui", fontSize: TEXT.size, wordWrap: true, wordWrapWidth: TEXT.wrap, align: "center" } });
    this.root.addChild(this.shape, this.words);
    this.root.zIndex = 50;
    this.root.visible = false;
    layer.addChild(this.root);
  }

  get text(): string | null {
    return this.current;
  }

  think(text: string, seconds: number): void {
    this.show(text, seconds, false);
  }

  say(text: string, seconds: number): void {
    this.show(text, seconds, true);
  }

  private show(text: string, seconds: number, spoken: boolean) {
    this.current = text;
    this.spoken = spoken;
    this.life = seconds;
    this.words.text = text;
    this.words.tint = spoken ? SPOKEN : 0x1a1a1a;
  }

  update(dt: number, at: { crown: Point; face: Rect; facing: 1 | -1; view: Rect }): void {
    this.life = Math.max(0, this.life - dt);
    if (this.life <= 0) this.current = null;
    this.root.visible = this.current !== null;
    if (!this.current) return;
    const size = { w: this.words.width + 2 * TEXT.padX, h: this.words.height + 2 * TEXT.padY };
    const { box, puffs } = bubbleLayout(size, at.crown, at.facing, at.face, at.view);
    const g = this.shape.clear();
    if (this.spoken) {
      g.roundRect(box.x, box.y, box.w, box.h, 14).fill(PAPER).stroke({ width: 4, color: INK });
    } else {
      // Four lobes across the box, then the box's middle filled over their inner edges.
      const r = box.h * 0.42;
      for (let i = 0; i < 4; i++) g.ellipse(box.x + box.w * (0.16 + 0.226 * i), box.y + box.h * (i % 2 ? 0.58 : 0.42), box.w * 0.2, r);
      g.fill(PAPER).stroke({ width: 4, color: INK });
      g.roundRect(box.x + box.w * 0.08, box.y + box.h * 0.18, box.w * 0.84, box.h * 0.64, 12).fill(PAPER);
      for (const p of puffs) g.circle(p.x, p.y, p.r).fill(PAPER).stroke({ width: 3, color: INK });
    }
    this.words.position.set(box.x + TEXT.padX, box.y + TEXT.padY);
    this.root.alpha = Math.min(1, this.life / FADE);
  }
}
```

```ts
/** The props between the last two steps, matched by id: what slow motion shows between them. */
export function between(prev: Snapshot, cur: Snapshot, alpha: number): Pick<Snapshot, "targets" | "snot" | "puddles"> {
  const at = (a: number, b: number) => a + (b - a) * alpha;
  const carts = new Map(prev.targets.map((t) => [t.id, t]));
  const blobs = new Map(prev.snot.map((b) => [b.id, b]));
  return {
    targets: cur.targets.map((t) => {
      const p = carts.get(t.id);
      return p ? { ...t, x: at(p.x, t.x), y: at(p.y, t.y) } : t;
    }),
    snot: cur.snot.map((b) => {
      const p = blobs.get(b.id);
      return p ? { ...b, x: at(p.x, b.x), y: at(p.y, b.y) } : b;
    }),
    puddles: cur.puddles,
  };
}
```

```ts
/** A particle's launch: speed and direction drawn from the recipe; `mirror` throws it the other way. */
export function launch(recipe: Recipe, a: number, b: number, mirror: boolean): { vx: number; vy: number } {
  const speed = recipe.speed[0] + (recipe.speed[1] - recipe.speed[0]) * a;
  const deg = recipe.angle[0] + (recipe.angle[1] - recipe.angle[0]) * b;
  const r = ((mirror ? 180 - deg : deg) * Math.PI) / 180;
  return { vx: Math.cos(r) * speed, vy: Math.sin(r) * speed };
}

/** One step of a particle's flight: gravity, drag, motion. */
export function drift(p: { x: number; y: number; vx: number; vy: number }, recipe: Recipe, dt: number): void {
  p.vy += recipe.gravity * dt;
  const k = Math.max(0, 1 - recipe.drag * dt);
  p.vx *= k;
  p.vy *= k;
  p.x += p.vx * dt;
  p.y += p.vy * dt;
}
```
